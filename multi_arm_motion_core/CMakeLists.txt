cmake_minimum_required(VERSION 3.22)
project(multi_arm_motion_core)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(multi_arm_motion_interfaces REQUIRED)
find_package(moveit_core REQUIRED)
find_package(rclcpp REQUIRED)
add_library(${PROJECT_NAME} SHARED src/prediction.cpp src/distance.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} multi_arm_motion_interfaces moveit_core rclcpp)
install(TARGETS ${PROJECT_NAME} EXPORT targets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(targets HAS_LIBRARY_TARGET)
ament_export_dependencies(multi_arm_motion_interfaces moveit_core rclcpp)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_prediction test/test_prediction.cpp)
  target_link_libraries(test_prediction ${PROJECT_NAME})
  find_package(urdf REQUIRED)
  find_package(srdfdom REQUIRED)
  ament_add_gtest(test_distance test/test_distance.cpp)
  target_link_libraries(test_distance ${PROJECT_NAME})
  ament_target_dependencies(test_distance urdf srdfdom)
endif()
ament_package()
