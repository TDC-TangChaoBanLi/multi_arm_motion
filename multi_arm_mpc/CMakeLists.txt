cmake_minimum_required(VERSION 3.22)
project(multi_arm_mpc)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(multi_arm_motion_core REQUIRED)
find_package(moveit_hybrid_planning REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_srvs REQUIRED)
add_library(mpc_solver SHARED src/solver.cpp src/braking.cpp)
target_compile_features(mpc_solver PUBLIC cxx_std_17)
target_include_directories(mpc_solver PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(mpc_solver multi_arm_motion_core)
add_library(mpc_local_solver SHARED src/local_solver.cpp)
target_link_libraries(mpc_local_solver mpc_solver)
ament_target_dependencies(mpc_local_solver moveit_hybrid_planning multi_arm_motion_core pluginlib std_srvs)
install(TARGETS mpc_solver mpc_local_solver EXPORT targets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
pluginlib_export_plugin_description_file(moveit_hybrid_planning plugins.xml)
ament_export_targets(targets HAS_LIBRARY_TARGET)
ament_export_dependencies(multi_arm_motion_core moveit_hybrid_planning)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_mpc test/test_mpc.cpp)
  target_link_libraries(test_mpc mpc_solver)
endif()
ament_package()
