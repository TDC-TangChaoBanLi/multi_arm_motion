cmake_minimum_required(VERSION 3.22)
project(multi_arm_hybrid_planning)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(moveit_hybrid_planning REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(multi_arm_motion_core REQUIRED)
find_package(multi_arm_mpc REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_srvs REQUIRED)
add_library(multi_arm_trajectory_operator SHARED src/trajectory_operator.cpp)
target_compile_features(multi_arm_trajectory_operator PUBLIC cxx_std_17)
ament_target_dependencies(multi_arm_trajectory_operator moveit_hybrid_planning pluginlib multi_arm_motion_core)
add_library(multi_arm_global_planner SHARED src/global_planner.cpp)
target_compile_features(multi_arm_global_planner PUBLIC cxx_std_17)
ament_target_dependencies(multi_arm_global_planner moveit_hybrid_planning pluginlib)
add_executable(command_router src/command_router.cpp)
target_compile_features(command_router PUBLIC cxx_std_17)
ament_target_dependencies(command_router rclcpp trajectory_msgs)
add_executable(hybrid_runtime src/hybrid_runtime.cpp)
target_compile_features(hybrid_runtime PUBLIC cxx_std_17)
ament_target_dependencies(hybrid_runtime moveit_hybrid_planning)
add_executable(coordinator src/coordinator.cpp)
target_compile_features(coordinator PUBLIC cxx_std_17)
ament_target_dependencies(coordinator multi_arm_motion_core multi_arm_mpc rclcpp_action std_srvs pluginlib)
target_link_libraries(coordinator multi_arm_mpc::mpc_solver)
install(TARGETS multi_arm_trajectory_operator multi_arm_global_planner LIBRARY DESTINATION lib)
install(TARGETS command_router hybrid_runtime coordinator RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
pluginlib_export_plugin_description_file(moveit_hybrid_planning plugins.xml)
pluginlib_export_plugin_description_file(moveit_hybrid_planning global_plugin.xml)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(urdf REQUIRED)
  find_package(srdfdom REQUIRED)
  ament_add_gtest(test_plugins test/test_plugins.cpp)
  ament_target_dependencies(test_plugins moveit_hybrid_planning pluginlib urdf srdfdom)
endif()
ament_package()
